                <div class="algorithm-right">
                    <div class="formula-box">
                        <div class="formula-title">Weighted Average Formulas</div>
                        <div class="formula">p<sub>ensemble</sub> = γp<sub>Mt,j</sub> + (1−γ)p<sub>y,i</sub></div>
                        <div class="formula">x<sub>ensemble</sub> = γx<sub>Mt,j</sub> + (1−γ)x<sub>y,i</sub></div>
                        <div class="formula">y<sub>ensemble</sub> = γy<sub>Mt,j</sub> + (1−γ)y<sub>y,i</sub></div>
                        <div class="formula">W<sub>ensemble</sub> = γW<sub>Mt,j</sub> + (1−γ)W<sub>y,i</sub></div>
                        <div class="formula">H<sub>ensemble</sub> = γH<sub>Mt,j</sub> + (1−γ)H<sub>y,i</sub></div>
                        <div class="formula-desc">Where γ ∈ [0, 1] controls the contribution of each model</div><div class="decoration"></div>
                    </div>
                    <div class="key-params">
                        <div class="param-box"><div class="param-icon"><i class="fas fa-sliders-h"></i></div><div class="param-title">Weighting Factor</div><div class="param-desc">γ = 0.5 assigns equal importance to both models</div><div class="decoration"></div></div>
                        <div class="param-box"><div class="param-icon"><i class="fas fa-filter"></i></div><div class="param-title">NMS Threshold</div><div class="param-desc">Removes boxes with IoU above threshold τ<sub>nms</sub></div><div class="decoration"></div></div>
                    </div>
                </div>
            </div>
        </div>
    </div>
